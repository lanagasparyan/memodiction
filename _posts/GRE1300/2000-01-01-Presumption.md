  word: Presumption

  - part_of_speech: Noun

    synonyms:

    antonyms:
    - Certainty
    - Proof

    synonyms:
    - Presumptuous
    - Arrogant
    - Overconfident
    antonyms:
    - Humble
    - Modest
    - Diffident

    origin: Middle English 'presumpcioun', Latin 'praesumptio' (a taking beforehand),

      century, evolving from Latin origins.
  idioms:
  - expression: '"Presumption of innocence"'
    explanation: The legal principle that a person is considered innocent unless proven

  regional_variations: []
  misconceptions:
